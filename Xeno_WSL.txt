wsl --import Xeno . .\rootfs.tar.gz 
wsl -d xeno --exec /bin/busybox --install -s /bin 
wsl --shutdown

# Connect VM
wsl -d Xeno

# Enabel APK
cd; mkdir -p /etc/apk
echo "http://dl-cdn.alpinelinux.org/alpine/latest-stable/main" > /etc/apk/repositories
echo "http://dl-cdn.alpinelinux.org/alpine/latest-stable/community" >> /etc/apk/repositories

wget https://dl-cdn.alpinelinux.org/alpine/latest-stable/main/x86_64/apk-tools-static-3.0.8-r0.apk
wget --no-check-certificate https://dl-cdn.alpinelinux.org/alpine/latest-stable/main/x86_64/apk-tools-static-3.0.8-r0.apk

tar xvzf apk-tools-static-3.0.8-r0.apk 
sbin/apk.static --initdb --allow-untrusted add apk-tools
apk add --allow-untrusted alpine-keys
apk update

rm -rf sbin .PKGINFO .SIGN.RSA.alpine-devel@lists.alpinelinux.org-6165ee59.rsa.pub apk-tools-static-3.0.8-r0.apk


# Timezone Setting
apk add tzdata
ln -sf /usr/share/zoneinfo/Asia/Seoul /etc/localtime
echo "Asia/Seoul" > /etc/timezone

# ZSH
apk add zsh zsh-autosuggestions zsh-syntax-highlighting 
cp /mnt/c/Alpine/config/.zshrc .

chsh or /etc/passwd  (/bin/zsh)
sed -i '/root/ s/sh/zsh/' /etc/passwd

# NeoVIM
apk add git curl nodejs neovim ripgrep build-base
mkdir .config
cp -R /mnt/c/Alpine/config/.config/nvim ~/.config 
nvim


# Remove VM
wsl --unregister xeno


# Make a rootfs
cd /mnt/c/Alpine
tar xvzf rootfs.tar.gz
cd roofs
wget https://dl-cdn.alpinelinux.org/alpine/edge/main/x86_64/busybox-static-1.38.0-r7.apk
tar xvzf busy*apk
mv bin/busybox.static busybox


tar -C rootfs -cvzf rootfs.tar.gz .




# NextVI static binaries
apk add build-base git
git clone https://github.com/kyx0r/nextvi
cd nextvi
CFLAGS="-static -O2 -s" ./cbuild.sh

cp /mnt/c/Alpine/bin/nextvi /bin
